SMART PILL DISPENSER - ALGORITHMS AND SOURCE CODE
=================================================
This document contains the exact Python code implementations for the core algorithms driving the Smart Pill Dispenser. It is designed to be used as a reference or appendix for research documentation.

1. THE MAIN DAEMON LOOP (Non-Blocking State Machine Algorithm)
--------------------------------------------------------------
The main loop operates as an asynchronous state machine. Instead of using `time.sleep()` which would block the CPU and paralyze the system, it continuously evaluates the time, sensors, and network flags hundreds of times per second.

```python
def main_loop():
    global is_dispense_window_active, active_compartment_ids, dispense_start_time
    
    # Start background sensor threads
    threading.Thread(target=temp_monitoring_loop, daemon=True).start()
    threading.Thread(target=sms_monitoring_loop, daemon=True).start()
    threading.Thread(target=telegram_polling_loop, daemon=True).start()
    
    last_dispensed_minute = None

    while True:
        now = datetime.now()
        
        # 1. Check Button Physical press
        if getattr(med_button, 'is_pressed', False):
            mark_medicine_taken()
            
        # 2. State Machine Window Countdown (5 Minutes)
        if is_dispense_window_active:
            if dispense_start_time and (now - dispense_start_time).total_seconds() > 300:
                # 5 minute timer expired
                db = SessionLocal()
                schedules = db.query(MedicationSchedule).filter(MedicationSchedule.compartment_id.in_(active_compartment_ids)).all()
                db.close()
                handle_missed_medication(schedules)
            time.sleep(0.1)
            continue

        # 3. Check Database for Scheduled Times
        current_time_str = now.strftime("%H:%M")
        if current_time_str != last_dispensed_minute:
            db = SessionLocal()
            current_date = now.date()
            schedules = db.query(MedicationSchedule).filter(
                MedicationSchedule.start_date <= current_date,
                MedicationSchedule.end_date >= current_date
            ).all()

            triggered = False
            for s in schedules:
                if current_time_str in [t.strip() for t in s.time_slots.split(",") if t.strip()]:
                    active_compartment_ids.append(s.compartment_id)
                    triggered = True

            if triggered:
                is_dispense_window_active = True
                dispense_start_time = now
                play_audio("alarm.wav")

            last_dispensed_minute = current_time_str
            db.close()
            
        time.sleep(0.1)
```

2. HARDWARE DISPENSING ALGORITHM (Servo Kinematics)
---------------------------------------------------
When the button is pressed inside an active window, the I2C PCA9685 controller translates the compartment requests into physical pulse-width-modulation (PWM) angles to drop the pills.

```python
def mark_medicine_taken():
    global is_dispense_window_active, active_compartment_ids, dispense_start_time
    
    if not is_dispense_window_active:
        return

    play_audio("dispensing.wav")
    
    # Rotate servo dynamically based on mechanical constraints
    try:
        for comp_id in active_compartment_ids:
            s = servos.get(comp_id)
            if s:
                if comp_id == 10:
                    s.angle = 60 # Drop pill
                    time.sleep(1) # Allow physical travel time
                    s.angle = 110 # Return to resting position
                elif comp_id == 4:
                    s.angle = 65
                    time.sleep(1)
                    s.angle = 110
                else:
                    s.angle = 70
                    time.sleep(1)
                    s.angle = 110
                    
                time.sleep(0.5) 
                s.angle = None  # Release the servo to stop electrical jittering
    except Exception as e:
        print(f"Servo error: {e}")

    # Reset State Machine
    is_dispense_window_active = False
    active_compartment_ids = []
    dispense_start_time = None
```

3. CLIMATE CONTROL ALGORITHM (Hysteresis & Cooldown)
----------------------------------------------------
This algorithm continuously polls the DHT11 sensor. It utilizes a software "Cooldown" flag to prevent the physical relay from flickering on/off rapidly when the temperature hovers precisely at 32.0 degrees, which would otherwise wear out the relay contacts.

```python
def temp_monitoring_loop():
    global current_temperature, cooling_active, cooling_mode, last_cooler_run_time
    cooler_on_duration = 120 # Run fan for 2 minutes minimum
    cooler_cooldown_interval = 1800 # Prevent turning on again for 30 minutes
    
    while True:
        now = datetime.now()
        
        # Auto mode logic
        if cooling_active:
            # If it is currently running, check if 2 minutes have passed
            if last_cooler_run_time and (now - last_cooler_run_time).total_seconds() >= cooler_on_duration:
                cooling_active = False
                if cooling_relay: cooling_relay.off()
        else:
            # If it is off, check if 30 minutes have passed since the last cooldown
            can_run = False
            if last_cooler_run_time is None:
                can_run = True
            else:
                if (now - last_cooler_run_time).total_seconds() >= cooler_cooldown_interval:
                    can_run = True
                    
            if can_run and current_temperature > 32.0:
                cooling_active = True
                last_cooler_run_time = now
                if cooling_relay: cooling_relay.on()
                    
        time.sleep(5)
```

4. TELEGRAM API LONG-POLLING ALGORITHM
--------------------------------------
This algorithm allows the Raspberry Pi to securely receive messages instantly from the internet without exposing itself to port-forwarding or dynamic DNS configuration, by keeping an outgoing request "hung" open for 10 seconds.

```python
def telegram_polling_loop():
    token = get_telegram_token()
    last_update_id = 0
    
    while True:
        try:
            # Use a 10-second timeout to hold the connection open (Long Polling)
            url = f"https://api.telegram.org/bot{token}/getUpdates?offset={last_update_id}&timeout=10"
            response = requests.get(url, timeout=15)
            
            if response.status_code == 200:
                data = response.json()
                if "result" in data:
                    for result in data["result"]:
                        # Shift the offset to mark the message as read
                        last_update_id = result["update_id"] + 1
                        
                        if "message" in result and "text" in result["message"]:
                            chat_id = str(result["message"]["chat"]["id"])
                            text = result["message"]["text"]
                            
                            # Prefix sender to distinguish Telegram from SMS in the backend
                            sender = "TG_" + chat_id
                            
                            # Send to Universal Command Parser
                            parse_sms_command(sender, text)
        except Exception as e:
            time.sleep(5)
```

5. SMS COMMUNICATION ALGORITHM (AT Commands)
--------------------------------------------
This algorithm allows the system to send and receive text messages without an internet connection by communicating directly with the SIM800L cellular module using the UART Serial interface and AT Commands.

```python
def send_sms(phone_number, message):
    try:
        # Open Serial Port to SIM800L
        ser = serial.Serial("/dev/ttyS0", 9600, timeout=1)
        
        # Set SIM to SMS Text Mode
        ser.write(b"AT+CMGF=1\r")
        time.sleep(0.5)
        
        # Initiate Send Command
        ser.write(f'AT+CMGS="{phone_number}"\r'.encode())
        time.sleep(0.5)
        
        # Write Message and End with CTRL+Z (ASCII 26)
        ser.write(message.encode() + b"\r")
        time.sleep(0.5)
        ser.write(bytes([26])) 
        time.sleep(3)
        ser.close()
    except Exception as e:
        print(f"Error sending SMS: {e}")

def sms_monitoring_loop():
    while True:
        try:
            ser = serial.Serial("/dev/ttyS0", 9600, timeout=1)
            ser.write(b"AT+CMGF=1\r")
            time.sleep(0.5)
            
            # Read all unread messages
            ser.write(b'AT+CMGL="REC UNREAD"\r')
            time.sleep(1)
            
            response = ser.read(ser.in_waiting).decode('utf-8', errors='ignore')
            lines = response.split('\n')
            
            # Delete messages from SIM memory after reading to prevent overflow
            ser.write(b"AT+CMGD=1,4\r")
            time.sleep(0.5)
            ser.close()
            
            # Parse SMS response block
            for i, line in enumerate(lines):
                if line.startswith("+CMGL:"):
                    parts = line.split(",")
                    if len(parts) >= 3:
                        sender = parts[2].replace('"', '')
                        # The next line contains the actual message text
                        if i + 1 < len(lines):
                            message = lines[i+1].strip()
                            parse_sms_command(sender, message)
                            
        except Exception as e:
            pass
            
        time.sleep(5)
```

6. UNIVERSAL REPLY ROUTER ALGORITHM (SMS + Telegram)
----------------------------------------------------
This algorithm is responsible for intercepting all outgoing text messages and dynamically deciding whether to send them via the internet (Telegram API) or via the cellular network (SIM800L), based purely on the structure of the recipient's ID.

```python
def send_sms(phone_number: str, message: str):
    if not phone_number:
        return
        
    # 1. INTERNET ROUTING (Telegram)
    # If the ID starts with "TG_", route to the Telegram API
    if phone_number.startswith("TG_"):
        chat_id = phone_number.replace("TG_", "")
        token = get_telegram_token()
        if token:
            url = f"https://api.telegram.org/bot{token}/sendMessage"
            try:
                requests.post(url, json={"chat_id": chat_id, "text": message})
            except Exception as e:
                pass
        return

    # 2. CELLULAR ROUTING (SIM800L)
    # Otherwise, split large messages into 160-character chunks
    # and send sequentially via AT commands.
    chunks = [message[i:i+150] for i in range(0, len(message), 150)]
    for i, chunk in enumerate(chunks):
        try:
            if len(chunks) > 1:
                chunk_data = (f"({i+1}/{len(chunks)}) " + chunk)[:160]
            else:
                chunk_data = chunk

            ser = serial.Serial("/dev/ttyS0", 9600, timeout=1)
            ser.write(b"AT+CMGF=1\r")
            time.sleep(0.5)
            ser.write(f'AT+CMGS="{phone_number}"\r'.encode())
            time.sleep(0.5)
            ser.write(chunk_data.encode() + b"\r")
            time.sleep(0.5)
            ser.write(bytes([26])) # CTRL+Z ends message
            time.sleep(3)
            ser.close()
            time.sleep(1) # Pause before sending next chunk
        except Exception as e:
            pass
```

7. UNIVERSAL COMMAND PARSER ALGORITHM
-------------------------------------
This massive algorithm acts as the central brain for processing remote user commands. It takes the text extracted by either the SMS or Telegram listeners and executes the corresponding hardware or database action.

```python
def parse_sms_command(sender: str, message: str):
    global cooling_mode, cooling_active
    message = message.strip()
    parts = message.split()
    if not parts: return

    db = SessionLocal()
    settings = db.query(SystemSettings).first()
    
    cmd = parts[0].lower()
    
    def reply_users():
        db.refresh(settings)
        send_sms(sender, f"User: {settings.user_mobile or 'None'}\nCare: {settings.caretakers or 'None'}")

    def reply_schedules():
        schs = db.query(MedicationSchedule).order_by(MedicationSchedule.compartment_id).all()
        if not schs:
            send_sms(sender, "No pills scheduled.")
        else:
            lines = ["Schedules:"]
            for s in schs:
                lines.append(f"C{s.compartment_id}: {s.medicine_name} [{s.time_slots}]")
            send_sms(sender, "\n".join(lines))

    # ---- COMMAND ROUTING EXAMPLES ---- #
    try:
        if cmd == "status":
            reply_users()
            reply_schedules()
            
        elif cmd == "cooling":
            if len(parts) > 1:
                mode = parts[1].lower()
                if mode in ["auto", "on", "off"]:
                    cooling_mode = mode
                    if mode == "on":
                        cooling_active = True
                        if cooling_relay: cooling_relay.on()
                    elif mode == "off":
                        cooling_active = False
                        if cooling_relay: cooling_relay.off()
                    send_sms(sender, f"Cooling mode set to {mode}.")
                    
        elif cmd == "addcare":
            # Link caretakers
            number = parts[1]
            if number.lower() == "me" and sender.startswith("TG_"):
                number = sender
            
            current = settings.caretakers or ""
            c_list = [c.strip() for c in current.split(",") if c.strip()]
            if number not in c_list:
                c_list.append(number)
                settings.caretakers = ",".join(c_list)
                db.commit()
                send_sms(sender, f"Added caretaker: {number}")
                
    except Exception as e:
        send_sms(sender, f"Error processing command: {str(e)}")
        
    db.close()
```

8. MISSED MEDICATION ESCALATION ALGORITHM
-----------------------------------------
This algorithm triggers if the patient fails to press the physical hardware button within the 5-minute dispensing window. It aggregates the list of missed medications and escalates the warning to the registered caretakers.

```python
def handle_missed_medication(schedules):
    global is_dispense_window_active, active_compartment_ids, dispense_start_time
    
    # 1. Reset hardware state machine
    is_dispense_window_active = False
    active_compartment_ids = []
    dispense_start_time = None

    # 2. Trigger audio alarm
    play_audio("missed_alert.wav")

    db = SessionLocal()
    settings = db.query(SystemSettings).first()
    db.close()
    
    if not schedules:
        return
        
    # 3. Aggregate list of missed pills
    names = ", ".join([s.medicine_name for s in schedules])
    slots = ", ".join([str(s.compartment_id) for s in schedules])

    alert_msg = f"ALERT: Patient missed scheduled medication. Slots: {slots}, Meds: {names}."
    user_msg = "Reminder: You missed your scheduled medication. Please take it immediately."

    # 4. Route SMS to user
    if settings.user_mobile:
        send_sms(settings.user_mobile, user_msg)
    
    # 5. Route SMS to all registered caretakers
    if settings.caretakers:
        for number in settings.caretakers.split(","):
            number = number.strip()
            if number:
                send_sms(number, alert_msg)
```
